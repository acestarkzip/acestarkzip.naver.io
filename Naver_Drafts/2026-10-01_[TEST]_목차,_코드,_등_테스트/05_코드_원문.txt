[4] PLAIN TEXT 코드
이미지 파일: images/이미지_04_code_plain_text_8f13fe7324.png

# 파일 권한 확인
ls -l test.sh

# 소유자에게 실행 권한 추가
chmod u+x test.sh

# 권한을 755로 변경
chmod 755 test.sh

# 권한을 644로 변경
chmod 644 test.txt

# 파일 소유자를 user로 변경
sudo chown user test.txt

# 파일 소유자와 그룹을 함께 변경
sudo chown user:group test.txt

[5] PLAIN TEXT 코드
이미지 파일: images/이미지_05_code_plain_text_d0aa8e3056.png

# 현재 사용자가 속한 그룹 확인
groups

# 사용자 및 그룹 정보 확인
id

# 새로운 그룹 생성
sudo groupadd robotics

# 사용자 user를 robotics 그룹에 추가
sudo usermod -aG robotics user

# 그룹 정보 확인
getent group robotics

# 그룹 삭제
sudo groupdel robotics

[7] PLAIN TEXT 코드

kill 1234

[9] PLAIN TEXT 코드
이미지 파일: images/이미지_09_code_plain_text_b139bb8208.png

# 패키지 목록 업데이트
sudo apt update

# 설치된 패키지 업데이트
sudo apt upgrade

# htop 패키지 설치
sudo apt install htop

# 설치된 패키지 정보 확인
apt show htop

# 설치된 패키지 삭제
sudo apt remove htop

# 패키지와 관련 설정 파일 삭제
sudo apt purge htop

# 더 이상 필요하지 않은 패키지 정리
sudo apt autoremove
