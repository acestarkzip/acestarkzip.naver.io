[5] PLAIN TEXT 코드
이미지 파일: images/이미지_05_code_plain_text_1a7bff9049.png

from abc import ABC, abstractmethod

class Robot(ABC):
    @abstractmethod
    def move(self):
        pass

class WheeledRobot(Robot):
    def move(self):
        print("바퀴로 이동합니다.")

robot = WheeledRobot()
robot.move()
