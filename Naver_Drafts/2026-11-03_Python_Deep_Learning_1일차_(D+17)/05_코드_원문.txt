import numpy as np
import matplotlib.pyplot as plt

x = np.linspace(-5, 5, 100)

# Sigmoid
sigmoid = 1 / (1 + np.exp(-x))

# ReLU
relu = np.maximum(0, x)

plt.plot(x, sigmoid, label="Sigmoid")
plt.plot(x, relu, label="ReLU")

plt.legend()
plt.grid()
plt.show()

import torch
import torch.nn as nn

class MLP(nn.Module):

    def __init__(self):
        super().__init__()

        self.network = nn.Sequential(
            nn.Linear(2, 4),
            nn.ReLU(),
            nn.Linear(4, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        return self.network(x)

model = MLP()

print(model)

import torch

# 기울기 계산을 추적할 Tensor
x = torch.tensor(2.0, requires_grad=True)

# 연산
y = x ** 2 + 3 * x + 1

# 역전파
y.backward()

# 기울기 출력
print("y:", y.item())
print("Gradient:", x.grad.item())
