#!/bin/bash

LOG_FILE="$HOME/system_check.log"

{
    echo "===== System Check ====="
    echo "Date: $(date)"
    echo "User: $(whoami)"

    echo ""
    echo "===== Disk Usage ====="
    df -h

    echo ""
    echo "===== Memory Usage ====="
    free -h

    echo ""
    echo "===== Check Complete ====="
} > "$LOG_FILE" 2>&1

echo "System check completed."
echo "Log saved to: $LOG_FILE"
