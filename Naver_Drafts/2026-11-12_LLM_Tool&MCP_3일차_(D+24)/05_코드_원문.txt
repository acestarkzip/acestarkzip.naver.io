import http.server
import socketserver
import webbrowser
import os
import tempfile

HTML_CONTENT = """
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <title>LLM Tool Calling: 3D 주사위 굴리기</title>
    <style>
        body { 
            font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; 
            display: flex; flex-direction: column; align-items: center; 
            justify-content: center; height: 100vh; background-color: #1e1e1e; 
            color: white; margin: 0; 
        }
        .scene { 
            width: 120px; height: 120px; perspective: 600px; margin: 40px; 
            pointer-events: none; /* 3D 요소가 클릭을 방해하지 못하게 차단 */
        }
        .cube { 
            width: 100%; height: 100%; position: relative; 
            transform-style: preserve-3d; transform: translateZ(-60px); 
            transition: transform 1.2s cubic-bezier(0.1, 0.8, 0.2, 1); 
        }
        .cube__face { 
            position: absolute; width: 120px; height: 120px; 
            background: white; border: 2px solid #ccc; border-radius: 12px; 
            display: flex; align-items: center; justify-content: center; 
            font-size: 50px; color: black; font-weight: bold; 
            box-shadow: inset 0 0 15px rgba(0,0,0,0.1); 
        }
        .cube__face--front  { transform: rotateY(  0deg) translateZ(60px); }
        .cube__face--right  { transform: rotateY( 90deg) translateZ(60px); }
        .cube__face--back   { transform: rotateY(180deg) translateZ(60px); }
        .cube__face--left   { transform: rotateY(-90deg) translateZ(60px); }
        .cube__face--top    { transform: rotateX( 90deg) translateZ(60px); }
        .cube__face--bottom { transform: rotateX(-90deg) translateZ(60px); }
        
        button { 
            position: relative; z-index: 999; /* 버튼을 화면 최상단으로 배치 */
            padding: 12px 24px; font-size: 16px; cursor: pointer; 
            background-color: #0078D4; color: white; border: none; 
            border-radius: 6px; font-weight: bold; transition: background 0.2s;
        }
        button:hover { background-color: #005a9e; }
        button:disabled { background-color: #555; cursor: not-allowed; }
        
        .log-box { 
            margin-top: 25px; padding: 20px; background: #2d2d2d; 
            border-radius: 8px; width: 320px; min-height: 200px; 
            white-space: pre-wrap; font-family: 'Consolas', monospace; 
            line-height: 1.5; box-shadow: 0 4px 6px rgba(0,0,0,0.3);
            position: relative; z-index: 998;
        }
    </style>
</head>
<body>
    <h2>[실습] 주사위 굴리기 도구</h2>
    <div class="scene">
        <div class="cube" id="cube">
            <div class="cube__face cube__face--front">1</div>
            <div class="cube__face cube__face--right">2</div>
            <div class="cube__face cube__face--back">3</div>
            <div class="cube__face cube__face--left">4</div>
            <div class="cube__face cube__face--top">5</div>
            <div class="cube__face cube__face--bottom">6</div>
        </div>
    </div>
    <button id="rollBtn">주사위 굴리기 (남은 횟수: 5)</button>
    <div class="log-box" id="logBox">대기 중...</div>

    <script>
        let rollCount = 0;
        const maxRolls = 5;
        let sum = 0;
        let rollHistory = [];
        
        const cube = document.getElementById('cube');
        const rollBtn = document.getElementById('rollBtn');
        const logBox = document.getElementById('logBox');

        const faceRotations = {
            1: 'rotateX(0deg) rotateY(0deg)',
            2: 'rotateX(0deg) rotateY(-90deg)',
            3: 'rotateX(0deg) rotateY(-180deg)',
            4: 'rotateX(0deg) rotateY(90deg)',
            5: 'rotateX(-90deg) rotateY(0deg)',
            6: 'rotateX(90deg) rotateY(0deg)'
        };

        function rollDice() {
            if (rollCount >= maxRolls) return;
            
            rollBtn.disabled = true;
            const result = Math.floor(Math.random() * 6) + 1;
            
            const xRand = Math.floor(Math.random() * 4 + 2) * 360;
            const yRand = Math.floor(Math.random() * 4 + 2) * 360;
            
            cube.style.transition = 'transform 1.2s cubic-bezier(0.1, 0.8, 0.2, 1)';
            cube.style.transform = `translateZ(-60px) rotateX(${xRand}deg) rotateY(${yRand}deg)`;

            setTimeout(() => {
                cube.style.transition = 'none';
                cube.style.transform = `translateZ(-60px) ${faceRotations[result]}`;
                
                sum += result;
                rollCount++;
                rollHistory.push(result);
                updateLog(result);
                
                if (rollCount < maxRolls) {
                    rollBtn.disabled = false;
                    rollBtn.innerText = `주사위 굴리기 (남은 횟수: ${maxRolls - rollCount})`;
                } else {
                    rollBtn.innerText = '굴리기 완료';
                    const historyStr = rollHistory.join(' + ');
                    logBox.innerText += `\\n이제 이 눈금들의 합을 계산하면:\\n[ ${historyStr} = ${sum} ]\\n\\n따라서 나온 눈금들의 합은 ${sum}입니다.`;
                }
            }, 1200);
        }

        function updateLog(result) {
            if (rollCount === 1) logBox.innerText = '5번 주사위를 굴린 결과는 다음과 같습니다:\\n\\n';
            logBox.innerText += `- ${rollCount}번째 굴림: ${result}\\n`;
        }

        // 인라인 onclick 대신 이벤트 리스너로 연결 (CSP 정책 우회)
        rollBtn.addEventListener('click', rollDice);
    </script>
</body>
</html>
"""

def serve_tool_ui():
    with tempfile.TemporaryDirectory() as temp_dir:
        html_path = os.path.join(temp_dir, "index.html")
        with open(html_path, "w", encoding="utf-8") as f:
            f.write(HTML_CONTENT)
        
        os.chdir(temp_dir)
        handler = http.server.SimpleHTTPRequestHandler
        
        with socketserver.TCPServer(("", 0), handler) as httpd:
            port = httpd.server_address[1]
            print(f"🎲 주사위 도구 UI가 시작되었습니다. (http://localhost:{port})")
            print("종료하려면 터미널에서 Ctrl+C를 누르세요.")
            
            webbrowser.open(f"http://localhost:{port}")
            try:
                httpd.serve_forever()
            except KeyboardInterrupt:
                print("\n서버를 종료합니다.")

if __name__ == "__main__":
    serve_tool_ui()

import os 
from openai import OpenAI
from dotenv import load_dotenv
import requests

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
client = OpenAI(api_key=api_key)

# 1. 날씨 가져오기
url_weather = "https://api.open-meteo.com/v1/forecast"
params_weather = {
    "latitude": 37.57,
    "longitude": 126.98,
    "current": "temperature_2m"
}

response_weather = requests.get(url_weather, params=params_weather)
print("상태 코드:", response_weather.status_code)
print("현재 온도:", response_weather.json()['current']['temperature_2m']) 


# 2. 경도 위도 찾아오기
# 실제 Geocoding API 엔드포인트 URL
url_geo = "https://geocoding-api.open-meteo.com/v1/search" 

# 도시명은 영문만 인식하므로 안내문 추가
city = input("위치 정보를 찾을 도시 이름을 영어로 입력하세요 (예: seoul): ") 

params_geo = {
    "name": city,
    "count": 1,
    "language": "en",
    "format": "json"
}

response_geo = requests.get(url_geo, params=params_geo)

# <Response [200]> 객체가 아닌 JSON 데이터 내용 확인
geo_data = response_geo.json()
print("\n지오코딩 응답 데이터:", geo_data)

# (참고) 만약 결과에서 위도와 경도만 쏙 뽑아내고 싶다면 아래처럼 할 수 있습니다.
if "results" in geo_data:
    lat = geo_data['results'][0]['latitude']
    lon = geo_data['results'][0]['longitude']
    print(f"찾은 위도: {lat}, 경도: {lon}")
else:
    print("도시를 찾을 수 없습니다.")
