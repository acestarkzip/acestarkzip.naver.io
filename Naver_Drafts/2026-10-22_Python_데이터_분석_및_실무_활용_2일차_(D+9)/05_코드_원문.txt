import pandas as pd

df = pd.DataFrame({
    "name": ["A", "B", "C"],
    "score": [80, 90, 75]
})

print(df.head())                 # 상위 5행 확인
print(df["score"].mean())        # 평균 계산
print(df[df["score"] >= 80])     # 조건에 맞는 행 조회

[2] PLAIN TEXT 코드
이미지 파일: images/이미지_02_code_plain_text_43d25ff98f.png

import pandas as pd

df = pd.DataFrame({
    "team": ["A", "A", "B", "B"],
    "score": [80, 90, 70, 100]
})

# 팀별 평균 점수
print(df.groupby("team")["score"].mean())

# 팀별 평균과 최댓값
print(df.groupby("team")["score"].agg(["mean", "max"]))
