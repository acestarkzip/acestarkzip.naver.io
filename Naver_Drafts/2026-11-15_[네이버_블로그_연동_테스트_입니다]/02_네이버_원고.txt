#include <iostream>
#include <Python.h> // 파이썬 헤더 파일 포함

int main() {
    // 1. 파이썬 인터프리터 초기화
    Py_Initialize();

    std::cout << "=== Python 코드 실행 테스트 ===" << std::endl;

    // 2. 간단한 파이썬 코드 문자열 실행
    PyRun_SimpleString("import sys");
    PyRun_SimpleString("print('Hello from Python! Python version:', sys.version)");
    PyRun_SimpleString("result = 10 + 20");
    PyRun_SimpleString("print('10 + 20 =', result)");

    // 3. 파이썬 인터프리터 종료
    Py_FinalizeEx();

    std::cout << "=== 테스트 종료 ===" << std::endl;
    return 0;
}
